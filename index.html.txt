        // Active Navigation Highlight on Scroll

        // Fetch YouTube Latest 3 Videos via RSS2JSON API
        const channelId = 'UCjuso5SChbtbYlcO7_j4SNQ';
        const rssUrl = `https://www.youtube.com/feeds/videos.xml?channel_id=${channelId}`;
        const apiUrl = `https://api.rss2json.com/v1/api.json?rss_url=${encodeURIComponent(rssUrl)}`;
        const videoGrid = document.getElementById('videoGrid');

        async function fetchYouTubeVideos() {
            try {
                const response = await fetch(apiUrl);
                const data = await response.json();

                if (data.status === 'ok' && data.items && data.items.length > 0) {
                    videoGrid.innerHTML = '';
                    // Get latest 3 videos
                    const latestVideos = data.items.slice(0, 3);

                    latestVideos.forEach(item => {
                        // Parse video ID from guid (e.g. yt:video:VIDEO_ID)
                        let videoId = '';
                        if (item.guid) {
                            const parts = item.guid.split(':');
                            videoId = parts[parts.length - 1];
                        }
                        if (!videoId && item.link) {
                            const urlParams = new URLSearchParams(new URL(item.link).search);
                            videoId = urlParams.get('v');
                        }

                        // Thumbnail fallback or standard high quality
                        const thumbnail = item.thumbnail || `https://img.youtube.com/vi/${videoId}/hqdefault.jpg`;
                        const title = item.title;
                        const pubDate = new Date(item.pubDate).toLocaleDateString('en-US', {
                            year: 'numeric',
                            month: 'short',
                            day: 'numeric'
                        });

                        const card = document.createElement('div');
                        card.className = 'video-card';
                        card.innerHTML = `
                            <div class="video-thumbnail-wrapper" style="cursor: pointer;" onclick="openVideoModal('${videoId}')">
                                <img src="${thumbnail}" alt="${title}" loading="lazy">
                                <div class="video-play-overlay">
                                    <i class="fa-solid fa-circle-play"></i>
                                </div>
                            </div>
                            <div class="video-details">
                                <h3 class="video-title">${title}</h3>
                                <div class="video-meta">
                                    <span><i class="fa-regular fa-calendar"></i> ${pubDate}</span>
                                    <a href="${item.link}" target="_blank" style="color: var(--primary); text-decoration: none; font-weight: 600;">YouTube <i class="fa-solid fa-arrow-up-right-from-square" style="font-size: 0.75rem;"></i></a>
                                </div>
                            </div>
                        `;
                        videoGrid.appendChild(card);
                    });
                } else {
                    throw new Error('No videos found');
                }
            } catch (error) {
                console.error('Error fetching YouTube RSS:', error);
                // Fallback direct static cards if CORS / API limit reached
                renderFallbackVideos();
            }
        }
